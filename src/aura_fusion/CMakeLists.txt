cmake_minimum_required(VERSION 3.8)
project(aura_fusion)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)

add_executable(fusion_node src/fusion_node.cpp)
target_compile_features(fusion_node PRIVATE cxx_std_17)
target_compile_options(fusion_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(fusion_node PRIVATE include)
target_link_libraries(fusion_node Eigen3::Eigen)
ament_target_dependencies(fusion_node rclcpp sensor_msgs nav_msgs rosgraph_msgs builtin_interfaces)
install(TARGETS fusion_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/record_comparison.py DESTINATION lib/${PROJECT_NAME})

# Keep your completed earlier exercise if it is already in your package.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/wheel_sensor_todo.cpp")

  target_compile_features(wheel_sensor_todo PRIVATE cxx_std_17)
  ament_target_dependencies(wheel_sensor_todo rclcpp sensor_msgs builtin_interfaces)
  install(TARGETS wheel_sensor_todo DESTINATION lib/${PROJECT_NAME})
endif()

if(BUILD_TESTING)
  enable_testing()
  add_executable(test_planar_ekf test/test_planar_ekf.cpp)
  target_compile_features(test_planar_ekf PRIVATE cxx_std_17)
  target_include_directories(test_planar_ekf PRIVATE include)
  target_link_libraries(test_planar_ekf Eigen3::Eigen)
  add_test(NAME planar_ekf_math COMMAND test_planar_ekf)
endif()
